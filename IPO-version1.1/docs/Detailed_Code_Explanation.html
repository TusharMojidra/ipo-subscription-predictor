from src.data_loader import load_data

# ─── Lines 2-3: Import ALL cleaning functions from preprocessing.py ───

                           fill_missing_values, winsorize_outliers)

from src.feature_engineering import add_subscription_class

def main():
    # ─── STEP 1: Load raw CSV data ───
    # load_data() reads the CSV file into a pandas DataFrame (a table)
    # "data/raw/ipo_data.csv" is the path to our raw dataset
    df = load_data("data/raw/ipo_data.csv")

    # Print how many rows (IPOs) and columns (features) we loaded
    print(f"Raw data loaded: {df.shape[0]} rows, {df.shape[1]} columns")
    # df.shape[0] = number of rows, df.shape[1] = number of columns

    # ─── STEP 2: Clean the data (4 sequential cleaning steps) ───
    df = clean_data(df)
    # → strips column names, parses dates, converts strings to numbers

    df = drop_unusable_rows(df)
    # → removes rows where the target column (Listing Gain) is missing
    # → removes rows where ALL subscription data (QIB/HNI/RII/Total) is missing

    df = fill_missing_values(df)
    # → fills remaining empty cells with the column's median value
    # → median is used instead of 0 because 0 would corrupt price data

    df = winsorize_outliers(df)
    # → caps extreme values at 1st and 99th percentile
    # → e.g., if one IPO had 500x subscription, it's capped to a sane value

    print(f"After cleaning: {df.shape[0]} rows")

    # ─── STEP 3: Create the classification target ───
    df = add_subscription_class(df)
    # → creates a new column "Subscription_Class" with values:
    #   "Low"    if Total subscription < 2x
    #   "Medium" if Total subscription is 2x–10x
    #   "High"   if Total subscription ≥ 10x

    # ─── STEP 4: Save cleaned data to disk ───
    df.to_csv("data/processed/cleaned_ipo_data.csv", index=False)
    # → writes the cleaned table to a new CSV file
    # → index=False means don't add a row-number column
    print("Data cleaning completed successfully!")

    # ─── STEP 5: Run Exploratory Data Analysis ───
    run_eda(df)
    # → generates charts: correlation heatmap, distributions, class balance
    # → saves them as PNG images in reports/figures/

    # ─── STEP 6: Train both prediction models ───
    run_classification(df)
    # → trains Logistic Regression, Random Forest, SVM for tier prediction
    # → saves results to reports/figures/classification_summary.csv

    run_regression(df)
    # → trains Huber, Linear, Decision Tree, RF, GBM for gain prediction
    # → saves results to reports/figures/regression_summary.csv

    # ─── Final message ───
    print("Pipeline finished. Cleaned data: data/processed/cleaned_ipo_data.csv")
    print("Figures & summaries: reports/figures/")


# ─── This ensures main() runs only when you execute this file directly ───
# (not when another file imports it)
if __name__ == "__main__":
    main()

🔑 Key Concepts in main.py

  • Sequential pipeline: Each step depends on the output of the previous one. Data flows in one direction.
  • def main(): The def keyword defines a function — a reusable block of code. main() is the standard name for the primary function.
  • f-strings (f"..."): Python's way of embedding variables inside strings. f"{df.shape[0]}" inserts the row count into the text.
  • if __name__ == "__main__": A Python convention — this block only runs when you execute python main.py, not when you import main.py from another file.

import pandas as pd

# ─── Line 2-3: Define the load_data function ───
# This function takes a file path (like "data/raw/ipo_data.csv")
# and returns a DataFrame with all the CSV data inside it
def load_data(path):
    # pd.read_csv() is pandas' built-in function to read CSV files
    # It automatically:
    #   - Reads the first row as column headers
    #   - Reads each subsequent row as data
    #   - Guesses data types (numbers, text, dates)
    return pd.read_csv(path)

🔑 Why is this a separate file?

Even though it's tiny, keeping it separate follows the Single Responsibility Principle — each file does ONE thing. If you later need to load data from an API or database instead of a CSV, you only change this one file.

src/preprocessing.py

Purpose: Contains 5 cleaning functions that transform raw, messy data into clean, model-ready data. Each function handles one specific type of messiness in the data.
src/preprocessing.pyPython
# ─── Lines 1-2: Import libraries ───
import pandas as pd    # For DataFrame operations
import numpy as np     # For numerical operations (NaN, quantile, clip)


# ═══════════════════════════════════════════════════════════════
# FUNCTION 1: remove_duplicates
# ═══════════════════════════════════════════════════════════════
def remove_duplicates(df):
    # drop_duplicates() removes rows that are exact copies of other rows
    # e.g., if the same IPO appears twice, one copy is removed
    return df.drop_duplicates()


# ═══════════════════════════════════════════════════════════════
# FUNCTION 2: clean_data — the main cleaning function
# ═══════════════════════════════════════════════════════════════
def clean_data(df):
    # Step A: Strip whitespace from column names
    # Sometimes CSV headers have trailing spaces like "Date " instead of "Date"
    # [c.strip() for c in df.columns] removes whitespace from every column name
    df.columns = [c.strip() for c in df.columns]

    # Step B: Convert the "Date" column from text to actual date objects
    # errors="coerce" means if a date can't be parsed, it becomes NaT (Not a Time)
    # format='mixed' handles different date formats in the same column
    df['Date'] = pd.to_datetime(df['Date'], errors="coerce", format='mixed')

    # Step C: Extract the year from each date and create a "Year" column
    # .dt.year accesses the year part of a datetime column
    df['Year'] = df['Date'].dt.year

    # Step D: Convert text numbers to actual numbers
    # These columns should be numeric but might contain text like "N/A" or spaces
    numeric_cols = ['Issue_Size(crores)', 'QIB', 'HNI', 'RII', 'Total',
                    'Offer Price', 'List Price', 'Listing Gain',
                    'CMP(BSE)', 'CMP(NSE)', 'Current Gains']

    # Loop through each column name in the list
    for c in numeric_cols:
        # Check if this column actually exists in our DataFrame
        if c in df.columns:
            # pd.to_numeric converts text to numbers
            # errors="coerce" turns unconvertible values into NaN (empty)
            df[c] = pd.to_numeric(df[c], errors="coerce")

    # Step E: Remove rows where the Date couldn't be parsed
    # ~df['Date'].isna() means "Date IS NOT empty"
    # The ~ symbol means "NOT" in pandas
    df = df[~df['Date'].isna()]

    return df


# ═══════════════════════════════════════════════════════════════
# FUNCTION 3: drop_unusable_rows
# ═══════════════════════════════════════════════════════════════
def drop_unusable_rows(df):
    # Step A: Remove rows where "Listing Gain" is missing (NaN)
    # This is our regression target — without it, we can't train or evaluate
    # .copy() creates a fresh copy so we don't modify the original data
    df = df[~df['Listing Gain'].isna()].copy()

    # Step B: Remove rows where ALL subscription data is missing
    # These 4 columns tell us how much demand the IPO had
    sub_cols = ['QIB', 'HNI', 'RII', 'Total']

    # .notna().all(axis=1) checks: "Are ALL 4 columns non-empty for this row?"
    # We KEEP only rows where ALL subscription data exists
    df = df[df[sub_cols].notna().all(axis=1)].copy()

    # Step C: Fix data error where List Price is exactly 0
    # An IPO can't list at ₹0 — this is a data entry error
    # We replace 0 with NaN (missing) so it gets handled by fill_missing_values
    df.loc[df['List Price'] == 0, 'List Price'] = np.nan

    # Reset the row index (0, 1, 2, ...) because we removed some rows
    # drop=True means don't add the old index as a column
    return df.reset_index(drop=True)


# ═══════════════════════════════════════════════════════════════
# FUNCTION 4: fill_missing_values
# ═══════════════════════════════════════════════════════════════
def fill_missing_values(df):
    # Step A: Find all columns that contain numbers
    # select_dtypes(include=[np.number]) returns only numeric columns
    numeric_cols = df.select_dtypes(include=[np.number]).columns

    # Step B: Fill empty cells with the column's median value
    # WHY median and not 0?
    #   - If Offer Price has a missing value and we fill it with 0,
    #     the model thinks that IPO was free — which corrupts the data
    #   - Median is the "middle value" — it represents the typical IPO
    #   - fillna(value) replaces all NaN/NaT cells with the given value
    for c in numeric_cols:
        df[c] = df[c].fillna(df[c].median())

    return df


# ═══════════════════════════════════════════════════════════════
# FUNCTION 5: winsorize_outliers
# ═══════════════════════════════════════════════════════════════
def winsorize_outliers(df, cols=None, lower=0.01, upper=0.99):
    # If no specific columns given, use these 5 columns
    # These are the columns most likely to have extreme outliers
    if cols is None:
        cols = ['Issue_Size(crores)', 'QIB', 'HNI', 'RII', 'Total']

    for c in cols:
        if c in df.columns:
            # quantile(0.01) = the value below which 1% of data falls
            # quantile(0.99) = the value below which 99% of data falls
            lo, hi = df[c].quantile(lower), df[c].quantile(upper)

            # clip(lo, hi) caps values: anything below lo becomes lo,
            # anything above hi becomes hi. This removes extreme outliers
            # while keeping the rest of the data unchanged.
            # Example: if HNI has a value of 999x but the 99th percentile
            # is 200x, it gets capped to 200x.
            df[c] = df[c].clip(lo, hi)

    return df

🔑 Why Winsorization?

One IPO might have 500x HNI subscription while the median is 15x. Without capping, this extreme value would dominate the model's training. Winsorization caps the tails without removing the data point entirely.

Data flow: Raw CSV (652 rows) → after cleaning (648 rows) — we lost 4 rows due to missing critical data.

src/feature_engineering.py

D:\IPO-version1.1\src\feature_engineering.py
Purpose: Creates new columns (features) from existing data that help the ML models make better predictions. The raw data alone isn't enough — we need to transform it into forms that reveal patterns.
src/feature_engineering.pyPython
# ─── Imports ───
import numpy as np      # For mathematical functions (log1p, NaN)
import pandas as pd    # For DataFrame operations

# ─── Constant: the 4 subscription columns ───
# These appear in multiple functions, so we define them once here
SUBSCRIPTION_COLS = ['QIB', 'HNI', 'RII', 'Total']


# ═══════════════════════════════════════════════════════════════
# FUNCTION 1: add_listing_gain
# ═══════════════════════════════════════════════════════════════
def add_listing_gain(df):
    # If the column already exists, don't recalculate it
    if 'Listing Gain' in df.columns:
        return df

    # Formula: ((Listing Price - Offer Price) / Offer Price) × 100
    # Example: Offer Price = ₹100, List Price = ₹150
    #          Listing Gain = ((150 - 100) / 100) × 100 = 50%
    df["Listing Gain"] = ((df["List Price"] - df["Offer Price"]) /
                          df["Offer Price"]) * 100
    return df


# ═══════════════════════════════════════════════════════════════
# FUNCTION 2: add_subscription_class — creates the classification target
# ═══════════════════════════════════════════════════════════════
def add_subscription_class(df, low=2.0, high=10.0):
    # This inner function classifies ONE IPO's subscription level
    def classify(x):
        if x < low:       # Total < 2x → "Low" (not many people applied)
            return 'Low'
        if x < high:      # 2x ≤ Total < 10x → "Medium" (moderate demand)
            return 'Medium'
        return 'High'      # Total ≥ 10x → "High" (everyone wants it!)

    # .apply(classify) runs the classify function on EVERY row's "Total" value
    # This creates a new column "Subscription_Class" with the result
    df['Subscription_Class'] = df['Total'].apply(classify)
    return df


# ═══════════════════════════════════════════════════════════════
# FUNCTION 3: add_log_features — the MOST IMPORTANT feature engineering
# ═══════════════════════════════════════════════════════════════
def add_log_features(df):
    # WHY log transform?
    # Subscription data is heavily right-skewed:
    #   - Most IPOs: 5x–30x subscription
    #   - Some IPOs: 200x+ subscription (extremely hot IPOs)
    # Without log: the 200x IPO dominates the model's training
    # With log: log1p(200) = 5.3, log1p(20) = 3.0 — much more balanced

    # np.log1p(x) = ln(x + 1). The +1 handles zeros (log(0) = undefined)
    for c in SUBSCRIPTION_COLS:
        df[f'log_{c}'] = np.log1p(df[c])
        # Creates: log_QIB, log_HNI, log_RII, log_Total

    # Also log-transform issue size and offer price
    # Issue sizes range from ₹10cr to ₹20,000cr — huge spread
    df['log_Issue_Size'] = np.log1p(df['Issue_Size(crores)'])
    df['log_Offer_Price'] = np.log1p(df['Offer Price'])

    return df


# ═══════════════════════════════════════════════════════════════
# FUNCTION 4: add_demand_ratios — relative demand features
# ═══════════════════════════════════════════════════════════════
def add_demand_ratios(df):
    # Total demand = sum of all 4 subscription columns
    # .replace(0, np.nan) prevents division by zero
    total_demand = df[SUBSCRIPTION_COLS].sum(axis=1).replace(0, np.nan)

    # QIB_share: What % of total demand came from institutional investors?
    # High QIB share → institutions are confident → likely good IPO
    df['QIB_share'] = df['QIB'] / total_demand

    # RII_share: What % came from retail investors?
    df['RII_share'] = df['RII'] / total_demand

    # HNI_to_QIB: Ratio of rich individual demand to institutional demand
    # High ratio → retail frenzy (might be overhyped)
    df['HNI_to_QIB'] = df['HNI'] / df['QIB'].replace(0, np.nan)

    # Issue_Price_Ratio: How big is the offering vs its price?
    # Large issue + high price = mega-cap IPO (different dynamics)
    df['Issue_Price_Ratio'] = df['Issue_Size(crores)'] / df['Offer Price'].replace(0, np.nan)

    return df


# ═══════════════════════════════════════════════════════════════
# FUNCTION 5: add_all_features — convenience wrapper
# ═══════════════════════════════════════════════════════════════
def add_all_features(df):
    # Applies BOTH log transforms AND demand ratios in one call
    # Nested function calls: first add_log_features runs, then add_demand_ratios
    return add_demand_ratios(add_log_features(df))

🔑 Why Log Transform? (Visual Example)

Before log: [5, 8, 12, 200, 350] — the 200 and 350 dominate
After log1p: [1.8, 2.2, 2.5, 5.3, 5.9] — much more balanced!

np.log1p(5) = ln(6) = 1.79
np.log1p(350) = ln(351) = 5.86
np.log1p(0) = ln(1) = 0 (safe — no errors from zeros!)

src/models_classification.py

D:\IPO-version1.1\src\models_classification.py
Purpose: Trains 3 different machine learning models to predict whether an IPO will be Low, Medium, or High demand. Uses an out-of-time split — trains on IPOs up to 2023, tests on 2024–2026 IPOs (simulating real future prediction).
src/models_classification.py (highlights)Python
# ─── Imports ───
import os
import matplotlib
matplotlib.use('Agg')  # Use non-interactive backend (no popup windows)
import matplotlib.pyplot as plt
import seaborn as sns
import pandas as pd

# The 3 ML models we'll compare
from sklearn.linear_model import LogisticRegression
from sklearn.ensemble import RandomForestClassifier
from sklearn.svm import SVC
# Metrics to evaluate model performance
from sklearn.metrics import (accuracy_score, classification_report,
                           confusion_matrix, f1_score,
                           precision_score, recall_score)
from sklearn.model_selection import GridSearchCV  # Auto-tunes hyperparameters
from sklearn.preprocessing import StandardScaler  # Normalizes features to same scale

# ─── Constants ───
# Only 3 features for classification — known BEFORE the IPO opens
# (We can't use subscription data because that IS the target we're predicting)
FEATURES = ['Issue_Size(crores)', 'Offer Price', 'Year']
TARGET = 'Subscription_Class'
CLASS_ORDER = ['Low', 'Medium', 'High']
SPLIT_YEAR = 2023  # Train on ≤2023, test on >2023


def run_classification(df, outdir="reports/figures", split_year=SPLIT_YEAR):
    # ─── Create output directory if it doesn't exist ───
    os.makedirs(outdir, exist_ok=True)

    # ─── Prepare X (inputs) and y (target) ───
    # .dropna() removes rows where any FEATURE or TARGET is missing
    data = df.dropna(subset=FEATURES + [TARGET]).copy()
    X = data[FEATURES]  # The 3 input columns
    y = data[TARGET]    # The "Low"/"Medium"/"High" column

    # ─── OUT-OF-TIME SPLIT ───
    # This is CRUCIAL — instead of random split, we split by YEAR:
    #   Train: all IPOs from 2010–2023
    #   Test:  all IPOs from 2024–2026
    # This simulates real prediction: can the model predict FUTURE IPOs?
    train_mask = data['Year'] <= split_year
    X_train, X_test = X[train_mask], X[~train_mask]
    y_train, y_test = y[train_mask], y[~train_mask]

    # ─── SCALING ───
    # StandardScaler normalizes all features to mean=0, std=1
    # WHY? Issue Size is in crores (100-5000), Offer Price in ₹ (10-2000),
    # Year is 2010-2026 — vastly different scales!
    # Without scaling, the model would think larger numbers = more important
    scaler = StandardScaler()
    X_train_s = scaler.fit_transform(X_train)  # Learn scale from training data
    X_test_s = scaler.transform(X_test)         # Apply same scale to test data

    # ─── DEFINE 3 MODELS WITH HYPERPARAMETER GRIDS ───
    models = {
        'Logistic Regression': (
            LogisticRegression(max_iter=2000, class_weight='balanced',
                               random_state=42),
            {'C': [0.1, 1, 10]}  # C controls regularization strength
        ),
        'Random Forest': (
            RandomForestClassifier(class_weight='balanced', random_state=42),
            {'n_estimators': [100, 200],  # How many trees?
             'max_depth': [None, 10]}     # How deep each tree?
        ),
        'SVM': (
            SVC(class_weight='balanced', random_state=42),
            {'C': [0.1, 1, 10],
             'kernel': ['rbf', 'linear']}  # Two different boundary shapes
        ),
    }

    # ─── TRAIN AND EVALUATE EACH MODEL ───
    for name, (model, grid) in models.items():
        # GridSearchCV tries every combination of hyperparameters
        # cv=5 means 5-fold cross-validation (tests on 20% of training data)
        gs = GridSearchCV(model, grid, cv=5, scoring='accuracy', n_jobs=-1)
        gs.fit(X_train_s, y_train)  # Train on training data

        best = gs.best_estimator_  # The best hyperparameter combo
        y_pred = best.predict(X_test_s)  # Predict on UNSEEN test data

        # Print detailed report
        print(classification_report(y_test, y_pred,
                                 target_names=CLASS_ORDER))

        # Generate confusion matrix plot
        cm = confusion_matrix(y_test, y_pred, labels=CLASS_ORDER)
        plt.figure(figsize=(6, 5))
        sns.heatmap(cm, annot=True, fmt='d', cmap='Blues',
                    xticklabels=CLASS_ORDER, yticklabels=CLASS_ORDER)
        plt.savefig(f"{outdir}replace(' ', '_').lower()}, dpi=150)
        plt.close()

    # ─── FEATURE IMPORTANCE ───
    # Random Forest naturally tells us which features matter most
    rf = RandomForestClassifier(n_estimators=200, class_weight='balanced',
                                random_state=42).fit(X_train_s, y_train)
    imp = pd.Series(rf.feature_importances_, index=FEATURES).sort_values(ascending=False)
    print("Random Forest feature importance:")
    print(imp.round(4).to_string())

🔑 Key Concepts

  • class_weight='balanced': Tells the model to pay extra attention to rare classes. Without this, the model would just predict "Medium" for everything (since Medium is most common) and get ~40% accuracy by ignoring Low and High entirely.
  • GridSearchCV: Instead of guessing the best settings, it tries ALL combinations and picks the one with highest cross-validation accuracy. Like trying every combination lock until one opens.
  • StandardScaler.fit_transform vs transform: fit_transform = "learn the scale + apply it" (training only). transform = "apply the SAME scale" (test data). Never fit on test data — that's cheating!
Why out-of-time split? A random 80/20 split would leak future information (a 2025 IPO in training helps predict a 2024 IPO). The time split ensures the model can ONLY use past data to predict the future — exactly like real life.

# ─── Imports ───
from sklearn.linear_model import HuberRegressor, LinearRegression
from sklearn.tree import DecisionTreeRegressor
from sklearn.ensemble import RandomForestRegressor, GradientBoostingRegressor
from sklearn.metrics import mean_absolute_error, mean_squared_error, r2_score
from src.feature_engineering import add_log_features

# 6 features — ALL log-transformed (includes subscription data)
# Unlike classification, here we CAN use subscription data
# because we're predicting AFTER bidding closes
FEATURES = ['log_QIB', 'log_HNI', 'log_RII', 'log_Total',
            'log_Issue_Size', 'log_Offer_Price']
TARGET = 'Listing Gain'
SPLIT_YEAR = 2023


def run_regression(df, outdir="reports/figures", split_year=SPLIT_YEAR):
    # ─── Add log features to the data ───
    df = add_log_features(df)

    # ─── Same out-of-time split pattern as classification ───
    data = df.dropna(subset=FEATURES + [TARGET]).copy()
    X, y = data[FEATURES], data[TARGET]
    train_mask = data['Year'] <= split_year
    X_train, X_test = X[train_mask], X[~train_mask]
    y_train, y_test = y[train_mask], y[~train_mask]

    # ─── Scale features ───
    scaler = StandardScaler()
    X_train_s = scaler.fit_transform(X_train)
    X_test_s = scaler.transform(X_test)

    # ─── 5 Regression Models ───
    models = {
        'Huber Regression': (
            HuberRegressor(max_iter=5000),
            {'alpha': [1e-3, 1e-2], 'epsilon': [1.35, 1.7]}
            # alpha = regularization strength
            # epsilon = how tolerant of outliers (higher = more robust)
        ),
        'Linear Regression': (LinearRegression(), {}),
        'Decision Tree': (
            DecisionTreeRegressor(random_state=42),
            {'max_depth': [5, 10, None]}
        ),
        'Random Forest': (
            RandomForestRegressor(random_state=42),
            {'n_estimators': [100, 200], 'max_depth': [None, 10]}
        ),
        'Gradient Boosting': (
            GradientBoostingRegressor(random_state=42),
            {'n_estimators': [100, 200], 'learning_rate': [0.05, 0.1]}
        ),
    }

    # ─── Train, evaluate, collect metrics ───
    for name, (model, grid) in models.items():
        if grid:
            gs = GridSearchCV(model, grid, cv=5,
                           scoring='neg_mean_squared_error', n_jobs=-1)
            gs.fit(X_train_s, y_train)
            best = gs.best_estimator_
        else:
            best = model.fit(X_train_s, y_train)

        y_pred = best.predict(X_test_s)

        # Three regression metrics:
        # RMSE: Root Mean Squared Error (lower = better, in same units as target)
        # MAE: Mean Absolute Error (average of |predicted - actual|)
        # R²: How much variance the model explains (1.0 = perfect, 0.0 = average)
        rmse = mean_squared_error(y_test, y_pred) ** 0.5
        mae = mean_absolute_error(y_test, y_pred)
        r2 = r2_score(y_test, y_pred)

        print(f"{name} — RMSE: {rmse:.3} | MAE: {mae:.3} | R²: {r2:.4}")

    # ─── Feature importance from best tree model ───
    for name, (model, _) in models.items():
        if name in ('Random Forest', 'Gradient Boosting'):
            imp = pd.Series(model.feature_importances_,
                           index=FEATURES).sort_values(ascending=False)
            # Save feature importance chart
            plt.figure(figsize=(8, 5))
            imp.plot(kind='barh')
            plt.savefig(f"{outdir}, dpi=150)

🔑 Why Huber Regression Wins

  • Linear Regression is sensitive to outliers — one extreme IPO can skew the whole model.
  • Huber Regression is a hybrid: it behaves like linear regression for normal points but like robust regression for outliers. Best of both worlds.
  • R² of ~0.30 seems low, but IPO listing gains are extremely noisy (influenced by market sentiment, news, GMP rumors). Explaining 30% of variance with just 6 numbers is actually decent.

MODEL_PATH = "models/ipo_models.joblib"

def _data_fingerprint():
    # Creates an MD5 hash of the cleaned CSV file
    # Used to detect if the data has changed since the models were cached
    with open("data/processed/cleaned_ipo_data.csv", "rb") as f:
        return hashlib.md5(f.read()).hexdigest()


def train_models(df, use_cache=True):
    fp = _data_fingerprint()

    # If cached models exist AND data hasn't changed → load from disk
    if use_cache and os.path.exists(MODEL_PATH):
        try:
            saved = joblib.load(MODEL_PATH)
            if saved.get("data_fp") == fp:
                return (saved["clf"], saved["scaler_clf"],
                        saved["reg"], saved["scaler_reg"])
        except Exception:
            pass  # Corrupt cache → retrain

    # ─── TRAIN CLASSIFIER ───
    df = add_log_features(df)
    clf = RandomForestClassifier(n_estimators=100, max_depth=4,
                                 class_weight='balanced', random_state=42)
    scaler_clf = StandardScaler()
    clf.fit(scaler_clf.fit_transform(df[CLASS_FEATURES]),
          df['Subscription_Class'])

    # ─── TRAIN REGRESSOR ───
    reg = GridSearchCV(HuberRegressor(max_iter=5000),
                       {'alpha': [1e-3, 1e-2], 'epsilon': [1.35, 1.7]},
                       cv=5, scoring='neg_mean_squared_error', n_jobs=-1)
    scaler_reg = StandardScaler()
    reg.fit(scaler_reg.fit_transform(df[GAIN_FEATURES]),
          df['Listing Gain'])
    reg = reg.best_estimator_  # Use the best hyperparameter combo

    # ─── CACHE TO DISK ───
    # joblib.dump saves the trained models as a binary file
    # Next time app.py starts, it loads this file instantly
    os.makedirs(os.path.dirname(MODEL_PATH) or ".", exist_ok=True)
    joblib.dump({
        "clf": clf, "scaler_clf": scaler_clf,
        "reg": reg, "scaler_reg": scaler_reg,
        "data_fp": fp
    }, MODEL_PATH)

    return clf, scaler_clf, reg, scaler_reg

🔑 The Caching System

First run: Train models (takes ~3-5 seconds) → save to models/ipo_models.joblib
Every subsequent run: Load from disk (instant!) → skip training
When data changes: MD5 hash no longer matches → auto-retrain

Why? The Streamlit dashboard would be slow if it retrained on every page refresh. Caching makes the app start in <1 second.

🔑 Streamlit Concepts You Need to Know

  • st.session_state: A dictionary that persists across page reruns. When the user types in an input, Streamlit reruns the entire script — session_state preserves values between reruns.
  • @st.cache_resource: A decorator that tells Streamlit to run this function ONCE and reuse the result forever (until app restart). Used for expensive operations like model training.
  • @st.cache_data: Same as cache_resource but for data (DataFrames, charts). Invalidates when inputs change.
  • st.markdown(..., unsafe_allow_html=True): Lets you inject raw HTML/CSS into the page. Used extensively for custom UI components.
  • st.html(...): Injects JavaScript that runs in the MAIN page (not an iframe). Used for cursor glow, zoom buttons, tooltip auto-dismiss.
  • st.iframe(...): Creates an isolated iframe for content. Used for the 3D CSS widget (needs its own scroll/sizing context).
  • unsafe_allow_javascript=True: Allows JavaScript execution. Required for st.html() with scripts.

🔑 The CSS Architecture

CSS Variables (--bg, --accent, etc.) are defined in :root with defaults. When a theme is selected, a second <style> block overrides ALL variables at once — the entire app recolors instantly without any JavaScript.

Font variables (--font-head, --font-body) work the same way. Changing the font picker updates the CSS variables, and every element in the app inherits the new fonts.

.stApp * { font-family: var(--font-body) !important; } — This aggressive selector forces ALL elements (including Streamlit's built-in widgets) to use the selected font.
